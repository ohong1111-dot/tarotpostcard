async function liveRender() { const sky = selSky.value; const ground = selGround.value; const exit = selExit.value; const figure = selFigure.value; const exitX = parseFloat(rngExitX.value); const exitY = parseFloat(rngExitY.value); const exitScale = parseFloat(rngExitScale.value); const exitRot = parseFloat(rngExitRot.value); const gW = parseFloat(rngGroundW.value); const gH = parseFloat(rngGroundH.value); const figX = parseFloat(rngFigX.value); const figY = parseFloat(rngFigY.value); const [rawSky, rawExit, rawGround, rawFigure] = await Promise.all([ loadImage(`/assets/skies/${sky}.png`), loadImage(`/assets/exits/${exit}.png`), loadImage(`/assets/grounds/${ground}.png`), loadImage(`/assets/figures/${figure}.png`) ]); // 天空、出口、地貌保留原始完整色調（不挖空亮部白光） const imgSky = rawSky; const imgExit = rawExit; const imgGround = rawGround; // 僅對旅人做邊緣白框防禦，若旅人也已去背乾淨可直接用 rawFigure const imgFigure = rawFigure; ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = '#e8e0cc'; ctx.fillRect(0, 0, canvas.width, canvas.height); const margin = 28; const innerW = canvas.width - margin * 2; const innerH = canvas.height - margin * 2; ctx.save(); ctx.beginPath(); ctx.rect(margin, margin, innerW, innerH); ctx.clip(); // 1. 天空滿版鋪底 if (imgSky) { ctx.save(); if (sky === 'sky_night') { ctx.translate(margin + innerW, margin); ctx.scale(-1, 1); ctx.drawImage(imgSky, 0, 0, innerW, innerH); } else { ctx.drawImage(imgSky, margin, margin, innerW, innerH); } ctx.restore(); } // 2. 出口（退後，石門中心白光完整保留不被鏤空） if (imgExit) { ctx.save(); const anchorX = margin + innerW * 0.80; const anchorY = margin + innerH * 0.75; ctx.translate(anchorX + exitX, anchorY + exitY); ctx.rotate((exitRot * Math.PI) / 180); ctx.scale(exitScale, exitScale); ctx.translate(-anchorX, -anchorY); ctx.drawImage(imgExit, margin, margin, innerW, innerH); ctx.restore(); } // 3. 地貌前景 if (imgGround) { ctx.save(); const groundDrawW = innerW * gW; const groundDrawH = innerH * gH; const groundDrawY = margin + innerH - groundDrawH; ctx.drawImage(imgGround, margin, groundDrawY, groundDrawW, groundDrawH); ctx.restore(); } // 4. 古典天光染色 if (SKY_TINTS && SKY_TINTS[sky]) { ctx.save(); ctx.globalCompositeOperation = 'multiply'; ctx.fillStyle = SKY_TINTS[sky]; ctx.fillRect(margin, margin, innerW, innerH); ctx.restore(); } // 5. 旅人姿態 if (imgFigure) { ctx.save(); const figH_px = innerH * 0.38; const figAspect = (imgFigure.naturalWidth / imgFigure.naturalHeight) || 0.6; const figW_px = figH_px * figAspect; const figPosX = margin + innerW * figX; const figPosY = margin + innerH * figY - figH_px; ctx.drawImage(imgFigure, figPosX, figPosY, figW_px, figH_px); ctx.restore(); } ctx.restore(); // 6. 偉特雙黑框 ctx.save(); ctx.strokeStyle = '#2b261f'; ctx.lineWidth = 4; ctx.strokeRect(margin, margin, innerW, innerH); ctx.lineWidth = 1.5; ctx.strokeRect(margin - 8, margin - 8, innerW + 16, innerH + 16); ctx.restore(); }
