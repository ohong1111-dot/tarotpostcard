// ---------------------------------------------------- // 透視校準 + 自動去白底 + 人物左下 / 出口右側銜接渲染引擎 // ---------------------------------------------------- async function renderLandscape(data) { const W = canvas.width; // 1200 const H = canvas.height; // 800 ctx.clearRect(0, 0, W, H); const [rawSky, rawGround, rawExit, rawFigure] = await Promise.all([ loadImage(`assets/skies/${data.sky}.png`), loadImage(`assets/grounds/${data.ground}.png`), loadImage(`assets/exits/${data.exit}.png`), loadImage(`assets/figures/${data.figure}.png`) ]); // 自動將白色背景轉為透明 const imgSky = rawSky; const imgGround = createTransparentCanvas(rawGround); const imgExit = createTransparentCanvas(rawExit); const imgFigure = createTransparentCanvas(rawFigure); // 1. 底紙羊皮紙 ctx.fillStyle = "#fdf9f0"; ctx.fillRect(0, 0, W, H); const margin = 32; const innerW = W - margin * 2; const innerH = H - margin * 2 - 36; ctx.save(); ctx.beginPath(); ctx.rect(margin, margin, innerW, innerH); ctx.clip(); // 版 1：天候色版（滿版鋪底） ctx.globalCompositeOperation = 'source-over'; if (imgSky) { ctx.drawImage(imgSky, margin, margin, innerW, innerH); } else { const skyGrad = ctx.createLinearGradient(0, margin, 0, margin + innerH * 0.7); skyGrad.addColorStop(0, ASSET_CONFIG.skies[data.sky].tone); skyGrad.addColorStop(1, "#fcf6e8"); ctx.fillStyle = skyGrad; ctx.fillRect(margin, margin, innerW, innerH); } // 版 2：立足地貌版（沉降至下半部，保留上方開闊天際） if (imgGround) { ctx.save(); ctx.globalCompositeOperation = 'source-over'; const gH = innerH * 0.52; const gY = margin + innerH - gH; ctx.drawImage(imgGround, margin, gY, innerW, gH); ctx.restore(); } // 版 3：遠方生路出口版（進一步靠右，基底嵌合於地貌稜線上） if (imgExit) { ctx.save(); ctx.globalCompositeOperation = 'source-over'; const exitAspect = (imgExit.width / imgExit.height) || 1.1; const exitW = innerW * 0.24; // 稍微收斂寬度，強化遠景縱深 const exitH = exitW / exitAspect; // 推向右側邊緣（X = 74% 處），底部嵌入中景地貌 const exitX = margin + (innerW * 0.74); const exitY = margin + (innerH * 0.76) - exitH; ctx.drawImage(imgExit, exitX, exitY, exitW, exitH); ctx.restore(); } // 版 4：旅人姿態黑線版（定錨於左下近景平台） if (imgFigure) { ctx.save(); ctx.globalCompositeOperation = 'source-over'; const isSit = data.figure.includes('sit'); const figH = isSit ? innerH * 0.28 : innerH * 0.38; const figAspect = (imgFigure.width / imgFigure.height) || 0.6; const figW = figH * figAspect; // 水平向左推至 8% 處，腳底貼齊近景下緣 const figX = margin + (innerW * 0.08); const figY = margin + (innerH * 0.90) - figH; ctx.drawImage(imgFigure, figX, figY, figW, figH); ctx.restore(); } ctx.restore(); // 結束裁切 // 1909 偉特古典雙層框線 ctx.strokeStyle = "#2b261f"; ctx.lineWidth = 3; ctx.strokeRect(margin, margin, innerW, innerH); ctx.lineWidth = 1; ctx.strokeRect(margin - 6, margin - 6, innerW + 12, innerH + 12); // 底部情境銘刻文字 ctx.fillStyle = "#3e382f"; ctx.font = "italic 20px 'Times New Roman', serif"; ctx.textAlign = "center"; const sceneTitle = `${ASSET_CONFIG.grounds[data.ground].name} · ${ASSET_CONFIG.skies[data.sky].name}`; ctx.fillText(`— ${sceneTitle} —`, W / 2, H - 18); }
